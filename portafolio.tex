% --- PERFIL CORPORATIVO ---
\begin{tcolorbox}[colback=white, colframe=SoidDark, arc=10pt, title=\textbf{QUIÉNES SOMOS}]
    SOID S.A.S. es una empresa líder en el sector logístico del Valle del Cauca, especializada en el transporte de carga extradimensionada y servicios de apoyo técnico vial. Nuestro compromiso radica en la eficiencia operativa y el estricto cumplimiento de las normas de seguridad.
\end{tcolorbox}

\begin{multicols}{2}
    \subsection*{Misión}
    Brindar soluciones logísticas integrales que aseguren el traslado exitoso de los activos de nuestros clientes, fundamentados en la innovación y la protección vial.
    
    \columnbreak
    
    \subsection*{Visión}
    Ser para el año 2030 el aliado estratégico número uno a nivel nacional en logística de maquinaria pesada y servicios de mensajería técnica.
\end{multicols}

% --- SERVICIOS ---
\section*{Servicios Especializados}

\begin{description}[style=unboxed,leftmargin=0cm]
    \item[\color{SoidGreen}\Large\textbf{01. CARGA EXTRADIMENSIONADA}] \hfill \\
    Traslado especializado de maquinaria agrícola (cosechadoras de caña, tractores, equipos industriales). Contamos con la logística necesaria para movimientos de gran escala en rutas nacionales.
    
    \item[\color{SoidOrange}\Large\textbf{02. ESCOLTA TÉCNICA VIAL}] \hfill \\
    Acompañamiento con vehículos certificados, equipados con pértigas y señalización reglamentaria. Gestión de permisos ante autoridades viales y estudios de ruta.
    
    \item[\color{SoidGreen}\Large\textbf{03. MENSAJERÍA Y ENCOMIENDAS}] \hfill \\
    Servicio de mensajería técnica ágil para el sector agroindustrial. Transporte de repuestos críticos y paquetería empresarial con entregas de alta prioridad.
\end{description}
